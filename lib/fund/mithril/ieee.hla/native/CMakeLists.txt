cmake_minimum_required(VERSION 3.16)
project(mithril_hla LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_path(OPENRTI_INCLUDE RTI/RTIambassador.h HINTS "${OPENRTI_ROOT}/include/rti1516e" REQUIRED)
find_library(OPENRTI_RTI rti1516e HINTS "${OPENRTI_ROOT}/lib" REQUIRED)
find_library(OPENRTI_TIME fedtime1516e HINTS "${OPENRTI_ROOT}/lib" REQUIRED)
add_executable(mithril-hla hla_host.cpp)
target_include_directories(mithril-hla PRIVATE "${OPENRTI_INCLUDE}")
target_link_libraries(mithril-hla PRIVATE "${OPENRTI_RTI}" "${OPENRTI_TIME}")
target_compile_definitions(mithril-hla PRIVATE RTI_DISABLE_WARNINGS)
